    isBase={isBase}

import ComponentTable from "../../components/ComponentTable.jsx";

export function isBase(page) {
    return [
        "Highlight",
        "HighlightContent",
        "MapIdentify",
        "ModalPopup",
    ].includes(page.title);
}

export function componentFile(page) {
    if (page.title === "InMapPopup") {
        return "HighlightPopup";
    } else {
        return page.title;
    }
}
